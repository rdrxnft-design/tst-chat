    <div id="chat-container"></div>

    <script>
        // 🌟 STEP A: CHANGE THIS TO YOUR EXACT TWITCH CHANNEL NAME
        const STREAMER_NAME = "MrManGuTTV";

        // 🌟 STEP B: ADD YOUR VIEWERS TO THEIR LEVELS (Type names in LOWERCASE)
        const viewerLevels = {
            "cocoHn": 1,
            "intocablezzzzz": 1,
            "mangumarketinggodai": 3,
            "super_supporter": 3,
            "another_user": 3
        };

        // This function triggers automatically every time someone types a message
        ComfyJS.onChat = (user, message, flags, self, extra) => {
            const container = document.getElementById("chat-container");
            const lowercaseUser = user.toLowerCase();

            // Check what level the viewer is. If they aren't on the list, give them Level 1 by default.
            let userLevel = viewerLevels[lowercaseUser] || 1;

            // Create a new message block
            const messageElement = document.createElement("div");
            messageElement.className = "chat-message";

            // Determine which custom badge image to load based on their level
            let badgeSrc = "";
            if (userLevel === 1) badgeSrc = "https://imgur.com/a/JaTp76o";
            if (userLevel === 2) badgeSrc = "https://imgur.com/q41sfgo";
            if (userLevel === 3) badgeSrc = "https://imgur.com/qskLuHA";

            // Build the internal look of the message (Badge + Name + Text)
            messageElement.innerHTML = `
                <img class="badge" src="${badgeSrc}" alt="Lvl ${userLevel}">
                <span class="username">${user}</span>
                <span class="message-text">${message}</span>
            `;

            // Append it to our chat screen
            container.appendChild(messageElement);

            // Automatically scroll down to show new messages
            window.scrollTo(0, document.body.scrollHeight);
        };

        // Connect the code to your live Twitch stream room
        ComfyJS.Init(STREAMER_NAME);
    </script>
